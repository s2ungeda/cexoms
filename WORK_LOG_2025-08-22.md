### Usage Example

```go
// Create account manager
accountManager := account.NewManager(keyManager, storageManager, nil)

// Create accounts
mainAccount, _ := accountManager.CreateAccount(ctx, &types.CreateAccountRequest{
    Name:     "main",
    Type:     types.AccountTypeMain,
    Exchange: "binance",
    // ... other fields
})

// Create transfer manager
transferManager := account.NewTransferManager(accountManager)

// Request transfer
transfer, _ := transferManager.RequestTransfer(ctx, &account.TransferRequest{
    FromAccount: "main",
    ToAccount:   "sub_arb",
    Asset:       "USDT",
    Amount:      decimal.NewFromInt(10000),
    Reason:      "Top up arbitrage account",
})

// Execute transfer
transferManager.ExecuteTransfer(ctx, transfer.ID)

// Set up automatic rebalancing
rebalancer := account.NewRebalancer(accountManager, transferManager, storage, config)
```
